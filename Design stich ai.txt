<script>
        function switchTab(tab) {
            const loginBtn = document.getElementById('tab-masuk');
            const registerBtn = document.getElementById('tab-daftar');
            const loginForm = document.getElementById('form-masuk');
            const registerForm = document.getElementById('form-daftar');

            if (tab === 'masuk') {
                // Button Styles
                loginBtn.classList.add('bg-white', 'text-primary', 'shadow-sm', 'ring-1', 'ring-black/5');
                loginBtn.classList.remove('text-on-surface-variant');
                registerBtn.classList.remove('bg-white', 'text-primary', 'shadow-sm', 'ring-1', 'ring-black/5');
                registerBtn.classList.add('text-on-surface-variant');

                // Form Transition
                loginForm.classList.remove('opacity-0', '-translate-x-full', 'absolute', 'pointer-events-none');
                loginForm.classList.add('opacity-100', 'translate-x-0');
                
                registerForm.classList.add('opacity-0', 'translate-x-full', 'absolute', 'pointer-events-none');
                registerForm.classList.remove('opacity-100', 'translate-x-0');
            } else {
                // Button Styles
                registerBtn.classList.add('bg-white', 'text-primary', 'shadow-sm', 'ring-1', 'ring-black/5');
                registerBtn.classList.remove('text-on-surface-variant');
                loginBtn.classList.remove('bg-white', 'text-primary', 'shadow-sm', 'ring-1', 'ring-black/5');
                loginBtn.classList.add('text-on-surface-variant');

                // Form Transition
                registerForm.classList.remove('opacity-0', 'translate-x-full', 'absolute', 'pointer-events-none');
                registerForm.classList.add('opacity-100', 'translate-x-0');
                
                loginForm.classList.add('opacity-0', '-translate-x-full', 'absolute', 'pointer-events-none');
                loginForm.classList.remove('opacity-100', 'translate-x-0');
            }
        }

        // Simple feedback for interaction
        document.querySelectorAll('form').forEach(form => {
            form.addEventListener('submit', (e) => {
                e.preventDefault();
                const btn = form.querySelector('button[type="submit"]');
                const originalText = btn.innerHTML;
                btn.disabled = true;
                btn.innerHTML = `<span class="animate-spin material-symbols-outlined">progress_activity</span> Memproses...`;
                
                setTimeout(() => {
                    btn.innerHTML = originalText;
                    btn.disabled = false;
                    alert('Simulasi: Akses Diterima. Mengalihkan ke Dashboard...');
                }, 1500);
            });
        });
    </script>

<script>
        let currentStep = 1;
        const totalSteps = 3;

        const nextBtn = document.getElementById('nextBtn');
        const prevBtn = document.getElementById('prevBtn');
        const progressLine = document.getElementById('progress-line');
        const dropzone = document.getElementById('dropzone');
        const fileInput = document.getElementById('fileInput');
        const fileList = document.getElementById('file-list');

        function updateStepUI() {
            // Update step content visibility
            for (let i = 1; i <= totalSteps; i++) {
                const content = document.getElementById(`step-content-${i}`);
                if (i === currentStep) {
                    content.classList.remove('hidden');
                    content.classList.add('animate-fade-in');
                } else {
                    content.classList.add('hidden');
                }
            }

            // Update step indicators
            document.querySelectorAll('.step-indicator').forEach((el, idx) => {
                const stepIdx = idx + 1;
                const circle = el.querySelector('.step-circle');
                const label = el.querySelector('span');

                if (stepIdx < currentStep) {
                    circle.innerHTML = '<span class="material-symbols-outlined">check</span>';
                    circle.classList.remove('bg-surface-container-highest', 'bg-primary');
                    circle.classList.add('bg-secondary', 'text-white');
                    label.classList.add('text-secondary');
                } else if (stepIdx === currentStep) {
                    circle.innerHTML = stepIdx;
                    circle.classList.remove('bg-surface-container-highest', 'bg-secondary');
                    circle.classList.add('bg-primary', 'text-white', 'ring-4', 'ring-primary/10');
                    label.classList.add('text-primary', 'font-bold');
                    label.classList.remove('text-on-surface-variant');
                } else {
                    circle.innerHTML = stepIdx;
                    circle.classList.remove('bg-primary', 'bg-secondary', 'ring-4');
                    circle.classList.add('bg-surface-container-highest', 'text-on-surface-variant');
                    label.classList.remove('text-primary', 'text-secondary', 'font-bold');
                    label.classList.add('text-on-surface-variant');
                }
            });

            // Update progress line
            const percentage = ((currentStep - 1) / (totalSteps - 1)) * 100;
            progressLine.style.width = `${percentage}%`;

            // Update buttons
            if (currentStep === 1) {
                prevBtn.classList.add('opacity-0', 'pointer-events-none');
            } else {
                prevBtn.classList.remove('opacity-0', 'pointer-events-none');
            }

            if (currentStep === totalSteps) {
                nextBtn.innerText = 'Ajukan Sekarang';
                nextBtn.classList.remove('bg-primary');
                nextBtn.classList.add('bg-secondary');
            } else {
                nextBtn.innerText = 'Lanjutkan';
                nextBtn.classList.add('bg-primary');
                nextBtn.classList.remove('bg-secondary');
            }
        }

        nextBtn.addEventListener('click', () => {
            if (currentStep < totalSteps) {
                currentStep++;
                updateStepUI();
            } else {
                alert('Peminjaman Berhasil Diajukan! Menunggu verifikasi admin.');
            }
        });

        prevBtn.addEventListener('click', () => {
            if (currentStep > 1) {
                currentStep--;
                updateStepUI();
            }
        });

        // Stepper Logic
        function increment(id) {
            const input = document.getElementById(id);
            const val = parseInt(input.value);
            if (val < parseInt(input.max)) input.value = val + 1;
        }

        function decrement(id) {
            const input = document.getElementById(id);
            const val = parseInt(input.value);
            if (val > 0) input.value = val - 1;
        }

        // Dropzone Logic
        dropzone.addEventListener('click', () => fileInput.click());
        
        dropzone.addEventListener('dragover', (e) => {
            e.preventDefault();
            dropzone.classList.add('border-primary', 'bg-primary/5');
        });

        dropzone.addEventListener('dragleave', () => {
            dropzone.classList.remove('border-primary', 'bg-primary/5');
        });

        dropzone.addEventListener('drop', (e) => {
            e.preventDefault();
            dropzone.classList.remove('border-primary', 'bg-primary/5');
            handleFiles(e.dataTransfer.files);
        });

        fileInput.addEventListener('change', (e) => {
            handleFiles(e.target.files);
        });

        function handleFiles(files) {
            Array.from(files).forEach(file => {
                const row = document.createElement('div');
                row.className = 'flex items-center justify-between p-3 bg-white border border-outline-variant/30 rounded-lg animate-fade-in';
                row.innerHTML = `
                    <div class="flex items-center gap-3">
                        <span class="material-symbols-outlined text-primary">description</span>
                        <div class="flex flex-col">
                            <span class="text-sm font-bold text-on-surface truncate max-w-[200px]">${file.name}</span>
                            <span class="text-xs text-on-surface-variant">${(file.size / 1024).toFixed(1)} KB</span>
                        </div>
                    </div>
                    <button type="button" class="text-error hover:bg-error/5 p-1 rounded transition-colors" onclick="this.parentElement.remove()">
                        <span class="material-symbols-outlined">delete</span>
                    </button>
                `;
                fileList.appendChild(row);
